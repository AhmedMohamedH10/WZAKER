  Click.xxx            <- صوت كل تسبيحة (لو مش موجود بيستخدم tasbeeh.wav)
  Completion.xxx       <- "بارك الله فيك": مرة واحدة عند الوصول للهدف (لو مش موجود بيستخدم complete.wav)
  Exit.xxx             <- "لسانك": عند الإغلاق النهائي فقط
  فسبحان الله.xxx      <- مرة واحدة عند بدء التشغيل
  Hover.xxx            <- صوت المرور على بطاقات التحدي (مرة عند دخول الماوس للبطاقة)
  Startup.xxx          <- نفس "فسبحان الله" (الاسم الإنجليزي أضمن؛ الاسم العربي القديم لسه بيشتغل)

  Reminder2.xxx        <- صوت الآية التانية في تنبيه الأذكار: "فسبحان الله حين تمسون وحين تصبحون" (الآيتين بيظهروا عشوائيًا)

  Birds.xxx / OWL.xxx  <- تنبيه أذكار الصباح / المساء
